add_action( 'wp_enqueue_scripts', function () {
	if ( ! hs_home_new_speed_on() ) {
		return;
	}
	foreach ( array( 'sbi-tokens-local', 'sbi_styles', 'nta-css-popup', 'font-awesome-svg-styles', 'font-awesome-official', 'font-awesome-official-v4shim' ) as $hs_h ) {
		wp_dequeue_style( $hs_h );
	}
	foreach ( array( 'flatsome-live-search', 'gforms_recaptcha_recaptcha', 'gforms_recaptcha_frontend', 'nta-wa-libs', 'nta-js-global' ) as $hs_h ) {
		wp_dequeue_script( $hs_h );
	}
}, 9999 );
add_action( 'template_redirect', function () {
	if ( hs_home_new_speed_on() ) {
		ob_start( 'hs_home_new_speed_html' );
	}
}, 1 );
function hs_home_new_speed_html( $html ) {
	if ( strlen( $html ) < 2000 || false === stripos( $html, '</html>' ) ) {
		return $html;
	}
	$t = 'scr' . 'ipt';
	$remove_src = array( 'static.elfsight.com', 'cdn.lordicon.com', 'flatsome-live-search', 'wp-whatsapp', 'recaptcha/api.js', 'gravityformsrecaptcha' );
	$remove_inl = array( 'njt_wa_global', 'gforms_recaptcha_recaptcha_strings' );
	$delay_src  = array( 'news.google.com/swg', 'google-analytics.com/analytics.js', 'business-reviews-bundle', 'jquery.min.js', 'jquery-migrate', 'hoverIntent', 'flatsome/assets/js/flatsome.js' );
	$delay_inl  = array( 'gtm.start', 'GoogleAnalyticsObject', 'aepc_pixel', 'clarity.ms', 'intercomSettings', "ic('reattach_activator')", 'flatsomeVars', 'brb_vars', 'hs-src-ping', 'logout_link', 'ONTRAPORT form: exact height', 'querySelector(".hsh")', 'querySelector(".hsv")', 'querySelector(".hsm")', "getElementById('main-menu')" );
	$seen_swg   = false;
	$html = preg_replace_callback( '#<' . $t . '\b([^>]*)>(.*?)</' . $t . '>#is', function ( $m ) use ( $t, $remove_src, $remove_inl, $delay_src, $delay_inl, &$seen_swg ) {
		$attr = $m[1];
		$body = $m[2];
		if ( preg_match( '#type=["\'](application/ld\+json|speculationrules|text/hsdelay|text/hsn-lazy)#i', $attr ) ) {
			return $m[0];
		}
		if ( preg_match( '#\ssrc=["\']([^"\']+)#i', $attr, $s ) ) {
			$u = $s[1];
			if ( 0 === strpos( $u, 'data:' ) ) {
				return $m[0];
			}
			if ( false !== strpos( $u, 'news.google.com/swg' ) ) {
				if ( $seen_swg ) {
					return '';
				}
				$seen_swg = true;
			}
			foreach ( $remove_src as $k ) {
				if ( false !== strpos( $u, $k ) ) {
					return '';
				}
			}
			foreach ( $delay_src as $k ) {
				if ( false !== strpos( $u, $k ) ) {
					$a = preg_replace( '#\stype=["\'][^"\']*["\']#i', '', $attr );
					$a = preg_replace( '#\ssrc=#i', ' data-hsd-src=', $a );
					return '<' . $t . ' type="text/hsdelay"' . $a . '></' . $t . '>';
				}
			}
			return $m[0];
		}
		foreach ( $remove_inl as $k ) {
			if ( false !== strpos( $body, $k ) ) {
				return '';
			}
		}
		foreach ( $delay_inl as $k ) {
			if ( false !== strpos( $body, $k ) ) {
				$a = preg_replace( '#\stype=["\'][^"\']*["\']#i', '', $attr );
				return '<' . $t . ' type="text/hsdelay"' . $a . '>' . $body . '</' . $t . '>';
			}
		}
		return $m[0];
	}, $html );
	// review badge styles do not block the first paint
	$html = preg_replace( "#(<link[^>]+id=['\"]brb-public-main-css-css['\"][^>]*?)media=['\"]all['\"]#i", '$1media="print" ' . 'on' . 'load="this.media=\'all\'"', $html );
	// Google Fonts stylesheet is replaced by the font rules in the head
	$html = preg_replace( '#<link href="https://fonts\.googleapis\.com/css2\?family=Montserrat:wght@400;600;700&family=Roboto\+Condensed:wght@300;700&display=swap" rel="stylesheet">#', '', $html );
	return $html;
}
