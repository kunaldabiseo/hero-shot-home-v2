
// Page 46735: Font Awesome is not used here; the review badge styles load without holding up the first paint
add_filter( 'style_loader_tag', function ( $tag, $handle ) {
	if ( ! hs_home_new_speed_on() ) {
		return $tag;
	}
	if ( 0 === strpos( $handle, 'font-awesome' ) ) {
		return '';
	}
	if ( 'brb-public-main-css' === $handle ) {
		return str_replace( array( "media='all'", 'media="all"' ), "media='print' data-hsn-async='1'", $tag );
	}
	return $tag;
}, 20, 2 );
// Page 46735: connect to the font server early, fetch the main font and the first hero photos first
add_filter( 'wp_resource_hints', function ( $urls, $type ) {
	if ( 'preconnect' === $type && is_page( 46735 ) ) {
		foreach ( $urls as $i => $u ) {
			if ( false !== strpos( is_array( $u ) ? (string) ( $u['href'] ?? '' ) : (string) $u, 'fontawesome' ) ) {
				unset( $urls[ $i ] );
			}
		}
		$urls[] = array( 'href' => 'https://fonts.gstatic.com', 'crossorigin' );
	}
	return $urls;
}, 10, 2 );
