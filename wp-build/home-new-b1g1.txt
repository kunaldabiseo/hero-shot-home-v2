
// Part 5: B1G1 impact numbers for the "give back" section on page 46735, read from B1G1's public widget feed (the same data their iframe widget shows). Cached 12 hours; the last good result is kept, so if B1G1 is down the page shows the last figures, and if there are none yet the written-in figures stay.
function hs_home_new_b1g1() {
	$cached = get_transient( 'hs_hn_b1g1_v1' );
	if ( is_array( $cached ) ) {
		return $cached;
	}
	$last = get_option( 'hs_hn_b1g1_last' );
	$last = is_array( $last ) ? $last : array();
	$res  = wp_remote_get( 'https://api.b1g1.com/widget/giving?key=f3c5001ce379c2d76c661f07acb3b1b0', array( 'timeout' => 8 ) );
	if ( is_wp_error( $res ) || 200 !== (int) wp_remote_retrieve_response_code( $res ) ) {
		set_transient( 'hs_hn_b1g1_v1', $last, HOUR_IN_SECONDS );
		return $last;
	}
	$j     = json_decode( wp_remote_retrieve_body( $res ), true );
	$total = isset( $j['impacts'] ) ? (int) $j['impacts'] : 0;
	$items = array();
	$projects = isset( $j['company']['company']['supported_projects'] ) && is_array( $j['company']['company']['supported_projects'] ) ? $j['company']['company']['supported_projects'] : array();
	foreach ( $projects as $p ) {
		$n = 0;
		if ( ! empty( $p['contributions'] ) && is_array( $p['contributions'] ) ) {
			foreach ( $p['contributions'] as $c ) {
				$n += isset( $c['beneficiaries'] ) ? (int) $c['beneficiaries'] : 0;
			}
		}
		if ( $n < 1 || empty( $p['phrase_active_plural'] ) ) {
			continue;
		}
		$items[] = array(
			'n'     => $n,
			'verb'  => isset( $p['verb_past'] ) ? (string) $p['verb_past'] : 'provided',
			'label' => 1 === $n && ! empty( $p['phrase_active_singular'] ) ? (string) $p['phrase_active_singular'] : (string) $p['phrase_active_plural'],
			'title' => isset( $p['project_title'] ) ? (string) $p['project_title'] : '',
		);
	}
	usort( $items, function ( $a, $b ) {
		return $b['n'] - $a['n'];
	} );
	$items = array_slice( $items, 0, 4 );
	if ( $total < 1 || count( $items ) < 1 ) {
		set_transient( 'hs_hn_b1g1_v1', $last, HOUR_IN_SECONDS );
		return $last;
	}
	$data = array( 'time' => time(), 'total' => $total, 'items' => $items );
	update_option( 'hs_hn_b1g1_last', $data, false );
	set_transient( 'hs_hn_b1g1_v1', $data, 12 * HOUR_IN_SECONDS );
	return $data;
}
